**Goal:** A tag push builds the zip and opens a DRAFT GitHub release whose body is player-facing notes written by the Claude API from the repo's change material; the user edits and publishes the draft on GitHub, and only then does the zip reach CurseForge (release body as its markdown changelog) and Discord (the repo's release webhook). Python unit tests run offline under a new `make tools-test` (part of `make check`).

**Tech Stack:** GitHub Actions, BigWigs packager v2, `gh` CLI, Python 3 standard library (urllib, json, re, subprocess, unittest), busted (Lua 5.1), Anthropic Messages API (`claude-opus-5-5`), CurseForge upload API.

```lua
-- tests/Forever/changelog_spec.lua (approved-release plan, docs/superpowers/plans/2026-10-19-approved-release.md):
-- CHANGELOG.md is the raw input of the generated release notes: an "## Unreleased" section first, carrying the five
-- headings implementers write under, and no released "## v..." section left empty.

local function read(path)
    local f = assert(io.open(path, "rb"), path)
    local s = f:read("*a")
    f:close()
    return s
end

-- { { name = "Unreleased", lines = { ... } }, ... } in file order; lines before the first "## " are the title
local function sections(text)
    local out, cur = {}, nil
    for line in (text .. "\n"):gmatch("([^\n]*)\n") do
        local name = line:match("^## (.-)%s*$")
        if name then
            cur = { name = name, lines = {} }
            out[#out + 1] = cur
        elseif cur then
            cur.lines[#cur.lines + 1] = line
        end
    end
    return out
end

local HEADINGS = { "New", "Changed", "Fixed", "Known issues", "Upgrade notes" }

describe("changelog", function()
    local all = sections(read("CHANGELOG.md"))

    it("opens with Unreleased, once", function()
        assert.are.equal("Unreleased", all[1] and all[1].name)
        for i = 2, #all do
            assert.are_not.equal("Unreleased", all[i].name)
        end
    end)

    it("gives Unreleased the five headings in order", function()
        local got = {}
        for _, line in ipairs(all[1].lines) do
            local h = line:match("^### (.-)%s*$")
            if h then
                got[#got + 1] = h
            end
        end
        assert.are.same(HEADINGS, got)
    end)

    it("names every other section after a tag and leaves none empty", function()
        assert.is_true(#all >= 2)
        for i = 2, #all do
            local s = all[i]
            assert.truthy(s.name:match("^v%d+%.%d+%.%d+"), s.name)
            local text = false
            for _, line in ipairs(s.lines) do
                if line:find("%S") and not line:match("^###") then
                    text = true
                end
            end
            assert.is_true(text, s.name .. " is empty")
        end
    end)
end)
```

```python
def fake_git(answers):
    """A git stand-in: answers[(args...)] is the stdout; a missing key raises like a failing git."""

    def run(*args):
        if args not in answers:
            raise subprocess.CalledProcessError(128, ["git", *args])
        return answers[args]

    return run


def repo(ref="v0.3.0", prev="v0.2.0", log="--- feat(hunter): pack\nbody line\nCo-Authored-By: X <x@y>\n"):
    describe = ("describe", "--tags", "--abbrev=0", "--match", "v[0-9]*", "--exclude", "*-*", ref + "^")
    answers = {
        ("log", "--no-merges", "--format=--- %s%n%b", prev + ".." + ref if prev else ref, "--", "src"): log,
        ("show", ref + ":CHANGELOG.md"): CHANGELOG,
        ("show", ref + ":docs/superpowers/plans/p.md"): PLAN,

        answers[describe] = prev + "\n"
        answers[("diff", "--name-only", "--diff-filter=AM", prev, ref, "--", rn.PLANS)] = "docs/superpowers/plans/p.md\n"
    return fake_git(answers)

class Response:
    def __init__(self, body):
        self.raw = io.BytesIO(json.dumps(body).encode())

    def read(self, *a):
        return self.raw.read(*a)

    def __enter__(self):
        return self

    def __exit__(self, *a):
        return False


class CallApi(unittest.TestCase):
    def call(self, body):
        seen = {}

        def opener(req, timeout):
            seen["req"], seen["timeout"] = req, timeout
            return Response(body)

        return rn.call_api({"model": "m"}, "sk-test", opener), seen

    def test_joins_text_blocks_and_skips_thinking(self):
        text, seen = self.call(
            {"stop_reason": "end_turn", "content": [{"type": "thinking", "thinking": ""}, {"type": "text", "text": " Notes "}]}
        )
        self.assertEqual(text, "Notes\n")
        req = seen["req"]
        self.assertEqual(req.full_url, rn.API_URL)
        self.assertEqual(req.get_header("X-api-key"), "sk-test")
        self.assertEqual(req.get_header("Anthropic-version"), "2023-06-01")
        self.assertEqual(json.loads(req.data), {"model": "m"})

    def test_refusal_truncation_and_empty_text_fail(self):
        for body in (
            {"stop_reason": "refusal", "content": []},
            {"stop_reason": "max_tokens", "content": [{"type": "text", "text": "half"}]},
            {"stop_reason": "end_turn", "content": [{"type": "text", "text": "  "}]},
        ):
            with self.assertRaises(SystemExit):
                self.call(body)

    def test_http_error_never_echoes_the_key(self):
        def opener(req, timeout):
            raise urllib.error.HTTPError(rn.API_URL, 401, "Unauthorized", {}, io.BytesIO(b'{"error":"bad key"}'))

        with self.assertRaises(SystemExit) as cm:
            rn.call_api({"model": "m"}, "sk-secret", opener)
        self.assertIn("401", str(cm.exception.code))
        self.assertNotIn("sk-secret", str(cm.exception.code))
```

```python
#!/usr/bin/env python3
"""Drafts the player-facing release notes for a tag (docs/publishing/README.md, "Cutting a release").

Gathers the change material between the previous release tag and <ref> (commits touching src/, the CHANGELOG.md
section for the tag or "Unreleased", and the "Open after" sections of plans changed since the previous tag), asks the
Anthropic Messages API to write the notes in the house style, and writes them to <out>.

usage: release_notes.py <ref> <out.md>       (ANTHROPIC_API_KEY in the environment)
       release_notes.py --material <ref>     (print the request's user message; no network)
Python 3 standard library only. Never prints the request headers or the key.
"""

import json
import os
import re
import subprocess
import sys
import urllib.error
import urllib.request

API_URL = "https://api.anthropic.com/v1/messages"
MODEL = "claude-opus-5-5"
PLANS = "docs/superpowers/plans"

SYSTEM = """\
You write the release notes players read for Hekili Forever, a World of Warcraft addon for the WoW Forever client \
that shows the next abilities to press. Players read the notes on CurseForge, GitHub and Discord.

The material is written by and for developers: changelog lines, commit messages and open-issue lists that name files, \
functions, tasks, test worlds, checkpoints and internal ids. Write only what a player would notice in game. Leave out \
anything a player cannot see (tests, tooling, refactors, documentation, internal models with no visible effect). Never \
write file, function or variable names, task, day, checkpoint or plan numbers, spell or spec ids, commit hashes, or \
internal terms such as belief, facade, harness, gate, oracle, family, hand, content API or pack API. Say each change \
in player terms: the class and specialization, the ability, and the situation where the recommendation or display \
now differs.

Format, GitHub markdown, under 60 lines in all:
- No title. Start with one sentence that sums up the release.
- One "### <Class>" section per class with changes, classes in alphabetical order, then "### Core and display" for \
changes every class sees. One bullet per change, one line each. Leave out empty sections.
- "### Known issues": the open issues a player could run into, one line each, written as what the player sees; leave \
out items that are only test gaps or developer to-dos. Leave the section out when there are none.
- "### Upgrade notes" only when the changelog's Upgrade notes say settings, profiles or saved priorities change; \
otherwise leave it out.
Do not invent changes. When the material does not make a change's effect on players clear, leave it out. Output only \
the notes."""


def git(*args):
    return subprocess.run(["git", *args], check=True, capture_output=True, text=True).stdout


def previous_tag(ref, run=git):
    """The newest non-prerelease v* tag before ref (so a final's notes span its release candidates), or None."""
    try:
        return run("describe", "--tags", "--abbrev=0", "--match", "v[0-9]*", "--exclude", "*-*", ref + "^").strip()
    except subprocess.CalledProcessError:
        return None


def section(text, heading):
    """The body under "## <heading>" up to the next "## ", stripped; "" when absent."""
    m = re.search(r"^## " + re.escape(heading) + r"[ \t]*\n(.*?)(?=^## |\Z)", text, re.M | re.S)
    return m.group(1).strip() if m else ""


def open_after(text):
    """Every "## Open after ..." section of a plan, heading included, joined by blank lines."""
    found = re.findall(r"^(## Open after[^\n]*\n.*?)(?=^## |\Z)", text, re.M | re.S)
    return "\n\n".join(s.strip() for s in found)


def gather(ref, run=git):
    prev = previous_tag(ref, run)
    span = prev + ".." + ref if prev else ref
    commits = run("log", "--no-merges", "--format=--- %s%n%b", span, "--", "src")
    commits = re.sub(r"^(Co-Authored-By|Claude-Session):.*\n?", "", commits, flags=re.M | re.I).strip()
    changelog = run("show", ref + ":CHANGELOG.md")
    entry = section(changelog, ref) or section(changelog, "Unreleased")
    known = []
    if prev:
        for path in run("diff", "--name-only", "--diff-filter=AM", prev, ref, "--", PLANS).split():
            body = open_after(run("show", ref + ":" + path))
            if body:
                known.append(body)
    return {"tag": ref, "previous": prev, "commits": commits, "changelog": entry, "known": "\n\n".join(known)}


def user_message(m):
    if not m["commits"] and not m["changelog"]:
        raise SystemExit("release_notes: no commits under src/ and no changelog entry since %s" % m["previous"])
    return (
        "Release %s (previous release: %s).\n\n"
        "<changelog>\n%s\n</changelog>\n\n"
        "<commits>\n%s\n</commits>\n\n"
        "<open_issues>\n%s\n</open_issues>\n\n"
        "Write the release notes for %s."
    ) % (m["tag"], m["previous"] or "none", m["changelog"], m["commits"], m["known"], m["tag"])


def build_request(m):
    return {
        "model": MODEL,
        "max_tokens": 16000,
        "output_config": {"effort": "high"},
        "fallbacks": "default",
        "system": SYSTEM,
        "messages": [{"role": "user", "content": user_message(m)}],
    }


def call_api(request, key, opener=urllib.request.urlopen):
    req = urllib.request.Request(
        API_URL,
        data=json.dumps(request).encode(),
        method="POST",
        headers={
            "content-type": "application/json",
            "x-api-key": key,
            "anthropic-version": "2023-06-01",
            "anthropic-beta": "server-side-fallback-2026-07-01",
        },
    )
    try:
        with opener(req, timeout=600) as resp:
            body = json.load(resp)
    except urllib.error.HTTPError as e:
        raise SystemExit("release_notes: API error %d: %s" % (e.code, e.read().decode(errors="replace")[:500]))
    if body.get("stop_reason") in ("refusal", "max_tokens"):
        raise SystemExit("release_notes: no notes (stop_reason %s)" % body["stop_reason"])
    text = "".join(b.get("text", "") for b in body.get("content", []) if b.get("type") == "text").strip()
    if not text:
        raise SystemExit("release_notes: the API returned no text")
    return text + "\n"


def main(argv):
    if len(argv) == 3 and argv[1] == "--material":
        print(user_message(gather(argv[2])))
        return 0
    if len(argv) != 3:
        raise SystemExit(__doc__)
    key = os.environ.get("ANTHROPIC_API_KEY")
    if not key:
        raise SystemExit("release_notes: ANTHROPIC_API_KEY is not set")
    notes = call_api(build_request(gather(argv[1])), key)
    with open(argv[2], "w") as f:
        f.write(notes)
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

```python
#!/usr/bin/env python3
"""Uploads a published GitHub release's zip to CurseForge (docs/publishing/README.md, "Cutting a release").

usage: cf_upload.py <release-event.json> <zip>      (CF_API_KEY in the environment)

The event is the workflow's $GITHUB_EVENT_PATH for `release: published`: the tag is the file's display name, the
release body (markdown) its changelog, and a prerelease uploads as a beta. Python 3 standard library only. Never
prints the token or the request.
"""

import json
import os
import sys
import urllib.error
import urllib.request
import uuid

API = "https://wow.curseforge.com/api"
PROJECT = 1753387
GAME_VERSION = "1.60.1"
FOREVER_TYPE = 88568  # CurseForge's game version type for WoW Forever (BigWigs packager release.sh, upload_curseforge)


def version_id(versions, name=GAME_VERSION, type_id=FOREVER_TYPE):
    ids = [v["id"] for v in versions if v.get("name") == name and v.get("gameVersionTypeID") == type_id]
    if len(ids) != 1:
        raise SystemExit("cf_upload: %d CurseForge game versions named %s of type %d" % (len(ids), name, type_id))
    return ids[0]


def metadata(release, game_version):
    body = (release.get("body") or "").strip()
    if not body:
        raise SystemExit("cf_upload: the release body is empty; edit the release notes, then publish again")
    return {
        "changelog": body,
        "changelogType": "markdown",
        "displayName": release["tag_name"],
        "gameVersions": [game_version],
        "releaseType": "beta" if release.get("prerelease") else "release",
    }


def multipart(meta, filename, data, boundary):
    """The multipart/form-data body CurseForge's upload-file takes: a `metadata` JSON field and the `file`."""
    b = boundary.encode()
    return b"".join(
        [
            b"--" + b + b"\r\n",
            b'Content-Disposition: form-data; name="metadata"\r\n',
            b"Content-Type: application/json\r\n\r\n",
            json.dumps(meta).encode(),
            b"\r\n--" + b + b"\r\n",
            b'Content-Disposition: form-data; name="file"; filename="' + filename.encode() + b'"\r\n',
            b"Content-Type: application/zip\r\n\r\n",
            data,
            b"\r\n--" + b + b"--\r\n",
        ]
    )


def call(url, token, data=None, content_type=None, opener=urllib.request.urlopen):
    headers = {"X-Api-Token": token}
    if content_type:
        headers["Content-Type"] = content_type
    req = urllib.request.Request(url, data=data, headers=headers, method="POST" if data is not None else "GET")
    try:
        with opener(req, timeout=300) as resp:
            return json.load(resp)
    except urllib.error.HTTPError as e:
        raise SystemExit("cf_upload: %s -> %d: %s" % (url, e.code, e.read().decode(errors="replace")[:500]))


def upload(release, zip_path, token, opener=urllib.request.urlopen):
    game_version = version_id(call(API + "/game/versions", token, opener=opener))
    meta = metadata(release, game_version)
    with open(zip_path, "rb") as f:
        data = f.read()
    boundary = uuid.uuid4().hex
    body = multipart(meta, os.path.basename(zip_path), data, boundary)
    url = "%s/projects/%d/upload-file" % (API, PROJECT)
    result = call(url, token, body, "multipart/form-data; boundary=" + boundary, opener)
    return meta, result


def main(argv):
    if len(argv) != 3:
        raise SystemExit(__doc__)
    token = os.environ.get("CF_API_KEY")
    if not token:
        raise SystemExit("cf_upload: CF_API_KEY is not set")
    with open(argv[1]) as f:
        release = json.load(f)["release"]
    meta, result = upload(release, argv[2], token)
    print("cf_upload: %s (%s) uploaded as CurseForge file %s" % (meta["displayName"], meta["releaseType"], result.get("id")))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

````markdown
## Cutting a release

**Standing rule for implementer briefs:** a commit that changes what a player sees adds its line under
`## Unreleased` in CHANGELOG.md (New / Changed / Fixed / Known issues / Upgrade notes) in the same commit.

A release is three steps: tag, read the draft, publish. Nothing reaches players before step 3. **A tag without a
published release is not released**: it has a draft on GitHub and nothing on CurseForge.

Tags are `vMAJOR.MINOR.PATCH`; a pre-release adds a suffix with a `-` (`v0.2.0-rc.1`, `v0.2.0-beta.1`) and goes to
CurseForge as a beta file.

1. **Tag.** On `forever` with `make check` green. For a final release, first rename `## Unreleased` in CHANGELOG.md to
   `## vX.Y.Z`, put a fresh `## Unreleased` with the five empty headings (`### New`, `### Changed`, `### Fixed`,
   `### Known issues`, `### Upgrade notes`) above it, and commit (a pre-release skips this; its notes read
   Unreleased). Then `git tag vX.Y.Z && git push origin vX.Y.Z` (that one tag, never `--tags`).
2. **Read the draft.** Actions > release (`.github/workflows/release.yml`) builds the zip with the BigWigs packager
   (no upload), writes the notes with `tools/release_notes.py` and creates a draft release with the zip attached
   (`Hekili-<tag>-forever.zip`; a pre-release for a `-` tag). The notes come from the changelog section for the tag
   (or Unreleased), the commits under `src/` since the previous release (pre-release tags are skipped, so a final
   release covers its release candidates) and the "Open after" sections of plans changed since then. Open
   Releases > the draft > Edit: fix anything that reads like a developer note, keep it player-facing, never leave the
   body empty.
3. **Publish.** Press Publish release (tick "Set as a pre-release" only if it already is one). Actions > publish
   (`.github/workflows/publish.yml`) uploads the attached zip to CurseForge project 1753387 with the release body as
   the markdown changelog, the tag as the file name and game version 1.60.1; the Discord webhook posts the release.

Edits after publishing reach GitHub only: fix the CurseForge changelog by hand on the file's page. Unpublishing or
deleting a release does not remove the CurseForge file.

To preview the notes without releasing: Actions > release > Run workflow, pick the branch or tag, keep `dry_run`
ticked. It builds the zip and prints the notes in the log (one Anthropic API call). Run from a tag with `dry_run`
unticked, it creates the draft (use this when a tag's run failed after the push, for example an API outage).

### Failure modes

- **Release run fails:** no draft exists. Read the log: `release_notes:` lines name the cause (missing key, API
  error, refusal, no changes since the last release). Fix it and re-run the job, or run the workflow on the tag with
  `dry_run` unticked. If a draft for the tag already exists, delete the draft first.
- **Publish run fails:** the release is public on GitHub (and the Discord webhook has posted it) but not on
  CurseForge. `cf_upload:` lines name the cause. Re-run the failed job once the cause is fixed
  (Actions > publish > Re-run).
- **CurseForge rejects a duplicate display name:** a tag can be uploaded once. To re-ship, cut a new tag (the next
  patch or `-rc.N`).
- **CurseForge daily upload limit:** the upload fails with a limit error; re-run the publish job the next day.
- **Game version missing:** `cf_upload: 0 CurseForge game versions named 1.60.1` means CurseForge has no such
  version under type 88568. When the client's `## Interface` changes, update `GAME_VERSION` in
  `tools/cf_upload.py`; list the versions with the curl below.

### One-time setup

- **Secrets** (GitHub repo > Settings > Secrets and variables > Actions): `CF_API_KEY` (CurseForge > API tokens; used
  only by publish.yml) and `ANTHROPIC_API_KEY` (console.anthropic.com > API keys; used only by release.yml). Actions
  masks both in logs; the tools never print a request or a header.
- **Discord webhook** (the user): in the channel's settings > Integrations, create a Discord webhook and copy its URL.
  Then GitHub repo > Settings > Webhooks > Add webhook: Payload URL is that Discord webhook URL with `/github`
  appended, content type `application/json`, "Let me select individual events", tick Releases only (untick Pushes).
  The Discord webhook posts each published release; the rehearsal confirms it stays quiet for drafts.

The CurseForge game version, with the same token:

```sh
curl -fsS -H "X-Api-Token: $CF_API_KEY" https://wow.curseforge.com/api/game/versions | jq '.[] | select(.name | test("^1\\.60")) | {id, name, gameVersionTypeID}'
```

does not push tags. The controller decides, and tells the user, whether the rehearsal release stays as a public beta
or is deleted (step 5).

1. Secrets and the Discord webhook are set up (above). Optional: Actions > release > Run workflow on `forever` with
   `dry_run` ticked; the log ends with the notes.
2. `git tag v0.1.1-rc.1 && git push origin v0.1.1-rc.1`. Actions > release goes green. Releases shows a draft
   pre-release `v0.1.1-rc.1` with `Hekili-v0.1.1-rc.1-forever.zip` attached. Read the notes: player language, one
   section per class then core and display, a Known issues list, under 60 lines, no file or task names. Nothing on
   CurseForge; no post from the Discord webhook.
3. Edit the notes if needed and publish. Actions > publish goes green (log: `cf_upload: v0.1.1-rc.1 (beta) uploaded

4. CurseForge project > Files shows `v0.1.1-rc.1`, type Beta, game version 1.60.1, its changelog equal to the release
   body. The Discord webhook has posted the release once.
5. Then, per the controller's decision: keep it as the public beta, or delete it: the file on CurseForge
   (Files > the file > Delete), then `gh release delete v0.1.1-rc.1 --cleanup-tag --yes` and
   `git tag -d v0.1.1-rc.1`.
````
