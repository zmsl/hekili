# Localization on AceLocale-3.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The contributor's `L[...]` wraps run on AceLocale-3.0 with one silent string table per addon (core `Hekili`,
each pack `Hekili_<Class>`), the packs' strings and zhCN translations live in the packs, debug output stays English,
curse names come from the client, built-in priority names show through a label, a lint catches unwrapped text, and the
English a player sees is byte-identical to the merge base.

**Architecture:** `src/Hekili/Libs/AceLocale-3.0` (vendored, Ace3 Release-r1403) loads through `embeds.xml`. Every addon
with text ships `Locales/enUS.lua` (generated by `tools/locales.py`, `NewLocale(app, "enUS", true, true)`) first in
its locale block, then one file per language. The core reads its table as `ns.L` (`Locales/Locales.lua`), a pack as
`Hekili:GetLocale("Hekili_<Class>")` (Content API 10). `Registry.PackLabel` shows a built-in priority's label from its
pack's table. A golden written at the merge base (`tests/Forever/fixtures/enus_text.txt`) pins enUS byte for byte.

**Tech Stack:** Lua 5.1 (WoW), AceLocale-3.0 (Ace3 tag `Release-r1403`, commit d295b12f8b889a30e86e0e901c5494df4b149c49),
busted and the headless harness (`tests/Game/harness/headless.lua`), Python 3 (`tools/locales.py`, unittest),
luacheck, StyLua, lua-language-server.

**Spec:** `docs/superpowers/specs/2026-10-25-localization-design.md` (binding; cited as "spec §N"). Line numbers below
are as of `forever-localization` 8d3863c90 (merge base with `forever`: 1ef1a63b).

## Review Focus

The five inputs most likely to bite a player that the spec implies but does not test; each has its test in the task
that owns the code.

1. **A curse the client has not cached at login** (`GetSpellInfo` answers nil): the setting shows the pack's English
   or translated string, never nil, and the saved choice (`elements`, a values key) is untouched. Test: Task 5 Step 1.
2. **A player's own priority, or one imported from a player on another language**: shown verbatim, never through a
   pack table, and the profile's `spec.package` stays the key whatever the label. Test: Task 6 Step 1.
3. **Two built-in priorities translated to the same text** (or a label equal to another's name): reads as a duplicate
   in the dropdown; `--check` fails. Test: Task 6 Step 5.
4. **A translator who put a pack's line in the core's file, or a pack language file created later**: the translation
   moves to the pack, never lost. Test: Task 3 Step 5 (`adopt`).
5. **A key no table holds** (text added after a translation shipped, any client): English, no error-handler call per
   miss (silent). Test: Task 2 Step 1.

local CLASSES = {

-- One line per text: backslash, line breaks and tabs escaped, and trailing spaces (an editor would trim them).
local function esc(s)
    s = s:gsub("\\", "\\\\"):gsub("\n", "\\n"):gsub("\t", "\\t")
    return (s:gsub(" +$", function(sp)
        return ("\\032"):rep(#sp)
    end))
end

-- A headless boot of class token `token` whose client answers M.SPELL_NAMES (fake_api answers "Spell <id>").
local function boot(token)
    local new = fake.new
    fake.new = function(...)
        local api, ctl = new(...)
        local info = api.C_Spell.GetSpellInfo
        api.C_Spell.GetSpellInfo = function(id)
            local i = info(id)
            if i and M.SPELL_NAMES[id] then
                i.name = M.SPELL_NAMES[id]
            end
            return i
        end
        return api, ctl
    end
    local ok, H = pcall(
        headless.new,
        { character = { class = token, classID = CLASSES[token][1], specID = CLASSES[token][2], level = 60 } }
    )
    fake.new = new
    assert(ok, H)
    assert(#H.loadErrors == 0, table.concat(H.loadErrors, "\n"))
    headless.cycle(H, H.ctl.now + 0.2)
    return H
end

-- Every name, desc and values string under `node`, as lines into `out` (a set).
local function walk(node, path, info, out, seen)
    if type(node) ~= "table" or seen[node] then
        return
    end
    seen[node] = true
    local function call(v)
        if type(v) ~= "function" then
            return v
        end
        local ok, r = pcall(v, info)
        return ok and r or nil
    end
    for _, field in ipairs({ "name", "desc" }) do
        local v = call(node[field])
        if type(v) == "string" and v ~= "" then
            out[path .. "\t" .. field .. "\t" .. esc(v)] = true
        end
    end
    local values = call(node.values)
    if type(values) == "table" then
        for k, v in pairs(values) do
            if type(v) == "string" then
                out[path .. "\tvalues." .. tostring(k) .. "\t" .. esc(v)] = true
            end
        end
    end
    local children = {}
    for k, v in pairs(type(node.args) == "table" and node.args or {}) do
        children[k] = v
    end
    for _, group in pairs(type(node.plugins) == "table" and node.plugins or {}) do
        for k, v in pairs(group) do
            children[k] = v
        end
    end
    for k, v in pairs(children) do
        local ci = {}
        for i, s in ipairs(info) do
            ci[i] = s
        end
        ci[#ci + 1] = k
        walk(v, path .. "." .. tostring(k), ci, out, seen)
    end
    seen[node] = nil
end

-- Calls fn(record) with Hekili:Print and Hekili:Notify recorded as `tag<TAB>text` lines.
local function capture(H, tag, out, fn)
    local hk = H.Hekili
    local print, notify = rawget(hk, "Print"), rawget(hk, "Notify")
    local function rec(_, s)
        out["~message\t" .. tag .. "\t" .. esc(tostring(s))] = true
    end
    hk.Print, hk.Notify = rec, rec
    pcall(fn)
    hk.Print, hk.Notify = print, notify
end

-- Spec.lua's RestoreDefaults notices for `kind` (updated, reverted, retired) at 1, 2 and 3 names.
local function restoreNotices(H, out)
    local hk, class = H.Hekili, H.Hekili.Class
    local p = hk.DB.profile
    local any
    for _, v in pairs(class.packs) do
        any = v
        break
    end
    local packs, retired = class.packs, class.retiredPacks
    for n = 1, 3 do
        local names = { "Alpha", "Beta", "Gamma" }
        for _, kind in ipairs({ "updated", "reverted", "retired" }) do
            local fresh = {}
            class.packs, class.retiredPacks = fresh, {}
            for i = 1, n do
                local name = names[i]
                if kind == "retired" then
                    class.retiredPacks[name] = true
                    p.packs[name] = { builtIn = true, lists = {} }
                else
                    fresh[name] = { version = any.version, import = any.import }
                    p.packs[name] = kind == "reverted" and { version = any.version + 1, builtIn = true, lists = {} }
                        or nil
                end
            end
            capture(H, kind .. " " .. n, out, function()
                hk:RestoreDefaults()
                H.ctl.now = H.ctl.now + 10
                H.RunTimers(H.ctl.now)
            end)
            for i = 1, n do
                p.packs[names[i]] = nil
            end
        end
    end
    class.packs, class.retiredPacks = packs, retired
end

-- The launcher text (UI.lua's GetDataText) for each mode, with and without the minor-cooldowns override.
local function launcher(H, out)
    local p = H.Hekili.DB.profile
    local mm = H.ns.UI.Minimap
    if not mm then
        return
    end
    for _, mode in ipairs({ "automatic", "single", "aoe", "dual", "reactive" }) do

            mm:RefreshDataText()
            out["~message\tlauncher " .. mode .. " " .. tostring(override) .. "\t" .. esc(mm.text)] = true
        end
    end
end

        H.Hekili.Engine.settings.screenshot = false -- the harness has no Screenshot()
        H.Hekili:MakeSnapshot()
    end)
    restoreNotices(H, out)
    launcher(H, out)
end

--- The sorted unique lines of every class pack's boot (addons.CONTENT, the shared content first, skipped).
function M.dump()
    local out = {}
    for _, name in ipairs(addons.CONTENT) do
        local token = name:match("^Hekili_(%a+)$"):upper()
        if CLASSES[token] then
            local H = boot(token)
            H.Hekili:RefreshOptions()
            walk(H.Hekili.Options, "options", {}, out, {})
            messages(H, out)
        end
    end
    contentNotices(out)
    local lines = {}
    for l in pairs(out) do
        lines[#lines + 1] = l
    end
    table.sort(lines)
    return lines
end

return M

def sync(path, text, check, stale):

def locales_dir(name):
    return os.path.join(SRC, name, "Locales")


def languages(name):
    """The addon's translation languages: its Locales/<locale>.lua files, enUS and Locales.lua aside."""
    d = locales_dir(name)
    if not os.path.isdir(d):
        return []
    return sorted(f[:-4] for f in os.listdir(d) if re.match(r"^[a-z]{2}[A-Z]{2}\.lua$", f) and f not in SKIP)


def adopt(keys, own, core, pack_only):
    """A pack language file's translations: its own, plus the core file's for its keys when the file is new (every
    one the core translates) or the key is the pack's alone (a line a translator put in the core's file)."""
    new = not own
    out = dict(own)
    for k in keys:
        if k not in out and k in core and (new or k in pack_only):
            out[k] = core[k]
    return out


def main(argv):
    check = "--check" in argv
    problems, stale, found = [], [], {}
    for name, toc in addons():
        found[name] = collect(toc)
        problems += found[name][1]
    core_keys = {k for keys in found[CORE][0].values() for k in keys}
    pack_only = {k for n, f in found.items() if n != CORE for keys in f[0].values() for k in keys} - core_keys
    # read before any file is rewritten: the packs adopt from them, and the core drops what a pack now holds
    core_tr = {loc: read_translations(os.path.join(locales_dir(CORE), loc + ".lua")) for loc in languages(CORE)}
    held = {}  # [locale] = the keys some pack's file of that language holds
    for name, (grouped, _) in sorted(found.items(), key=lambda f: f[0] == CORE):  # the packs first, then the core
        if not grouped:

        for locale in languages(name):

            translations = read_translations(path)

                for k in sorted(k for k in translations if k in pack_only):
                    if check:
                        problems.append(

                        )

# ---------------------------------------------------------------------------------------------------------------------
# The lint: a player-visible literal not inside L[...] (docs/dev/architecture.md, "Localization"; spec §7). Token-level,
# not a parser. Shortcut: a literal returned from a function in a sink (`name = function() return "x" end`) is not
# seen; add a rule only if such lines recur (review catches them).

SINK_FIELDS = {"name", "desc", "text", "label", "tooltipTitle", "tooltipText", "confirmText"}
SINK_CALLS = {"SetText": 1, "SetFormattedText": 1, "AddLine": 1, "Notify": 1, "Print": 1, "AddDoubleLine": 2}
LINT_SKIP = re.compile(r"/Libs/|/Locales/|^src/ForeverProbe/|^src/Hekili/Options/DevTools\.lua$|"
                       r"^src/Hekili/Engine/Snapshot\.lua$|^src/Hekili/Game/State/Snapshot\.lua$")
VERBATIM = "-- l10n: verbatim"
NOT_TEXT = re.compile(r"\|c[0-9A-Fa-f]{8}|\|r|\|T[^|]*\|t|\|A[^|]*\|a|%(?:\d+\$)?[-+ #0]*\d*(?:\.\d+)?[sdifgxXcq%]")


def tokens(src):
    """[(kind, value, line)] of a Lua source: 'str' (decoded), 'name', 'op'; comments and numbers dropped."""
    out, i, n, line, at = [], 0, len(src), 1, 0
    while i < n:
        j = skip_space(src, i)
        if j != i:
            i = j
            continue
        c = src[i]
        line, at = line + src.count("\n", at, i), i
        lit = scan_literal(src, i) if (c in "\"'" or re.match(r"\[=*\[", src[i:i + 64])) else None
        if lit:
            out.append(("str", lit[0], line))
            i = lit[1]
            continue
        m = re.match(r"[A-Za-z_]\w*|\d[\w.]*|\.\.\.?|[=~<>]=|::|.", src[i:])
        tok = m.group(0)
        if tok[0].isdigit():
            pass
        elif tok[0].isalpha() or tok[0] == "_":
            out.append(("name", tok, line))
        else:
            out.append(("op", tok, line))
        i += len(tok)
    return out


def is_text(s):
    return re.search(r"[A-Za-z]", NOT_TEXT.sub("", s)) is not None


def literal_at(toks, k):
    """(text, line, next index) of the literal or `..` chain of literals at toks[k], else None. Also takes a
    format string: format("...", ...), string.format("...", ...), ("..."):format(...)."""
    t = toks[k] if k < len(toks) else None
    if not t:
        return None
    if t[0] == "str":
        parts, line, k = [t[1]], t[2], k + 1
        while k + 1 < len(toks) and toks[k][1] == ".." and toks[k + 1][0] == "str":
            parts.append(toks[k + 1][1])
            k += 2
        return "".join(parts), line, k
    if t == ("name", "string", t[2]) and k + 2 < len(toks) and toks[k + 1][1] == "." and toks[k + 2][1] == "format":
        k += 2
        t = toks[k]
    if t[1] == "format" and k + 2 < len(toks) and toks[k + 1][1] == "(" and toks[k + 2][0] == "str":
        return toks[k + 2][1], toks[k + 2][2], k + 3
    if t[1] == "(" and k + 4 < len(toks) and toks[k + 1][0] == "str" and toks[k + 2][1] == ")" \
            and toks[k + 3][1] == ":" and toks[k + 4][1] == "format":
        return toks[k + 1][1], toks[k + 1][2], k + 5
    return None


def args_start(toks, k):
    """Indexes of the first tokens of the arguments of the call whose '(' is toks[k]."""
    starts, depth, j = [k + 1], 0, k + 1
    while j < len(toks):
        v = toks[j][1] if toks[j][0] == "op" else None
        if v in ("(", "{", "["):
            depth += 1
        elif v in (")", "}", "]"):
            if depth == 0:
                break
            depth -= 1
        elif v == "," and depth == 0:
            starts.append(j + 1)
        j += 1
    return starts


def violations(src, rel):
    """[(line, why)] of unwrapped player-visible literals in sinks (spec §7)."""
    toks, out = tokens(src), []
    lines = src.split("\n")

    def flag(hit, why):
        if hit and is_text(hit[0]) and not lines[hit[1] - 1].rstrip().endswith(VERBATIM):
            out.append((hit[1], "%s: %r is not wrapped in L[...]" % (why, hit[0][:60])))

    for k, t in enumerate(toks):
        prev = toks[k - 1][1] if k else ""
        if t[0] == "name" and t[1] in SINK_FIELDS and prev in ("{", ",", ";") and k + 1 < len(toks) and toks[k + 1][1] == "=":
            flag(literal_at(toks, k + 2), "field " + t[1])
        elif t[0] == "name" and t[1] == "values" and prev in ("{", ",", ";") and k + 2 < len(toks) \
                and toks[k + 1][1] == "=" and toks[k + 2][1] == "{":
            for s in args_start(toks, k + 2):
                j = s + 2 if s + 1 < len(toks) and toks[s][0] == "name" and toks[s + 1][1] == "=" else s
                if j < len(toks) and toks[j][1] == "[":  # ["key"] = value
                    while j < len(toks) and toks[j][1] != "=":
                        j += 1
                    j += 1
                flag(literal_at(toks, j), "values")
        elif t[0] == "name" and t[1] in SINK_CALLS and prev == ":" and k + 1 < len(toks) and toks[k + 1][1] == "(":
            for s in args_start(toks, k + 1)[: SINK_CALLS[t[1]]]:
                flag(literal_at(toks, s), ":" + t[1])
    return out


def lint():
    """Every unwrapped player-visible literal of every addon file (spec §7), as "file:line: why"."""
    problems = []
    for rel in lua_files():
        if LINT_SKIP.search(rel):
            continue
        with open(os.path.join(ROOT, rel), encoding="utf-8") as f:
            src = f.read()
        problems += ["%s:%d: %s" % (rel, line, why) for line, why in violations(src, rel)]
    if problems:
        problems.append("wrap each in L[...], or end its line with %s when it is on the never-localized list" % VERBATIM)
    return problems
